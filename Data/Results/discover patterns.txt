public static bool IsBTTSYes(MatchData match)
{
    // STEP 1: Core BTTS Trigger
    bool lowAverageGoals =
        match.HomeAvgGoals < 1.30 &&
        match.AwayAvgGoals < 1.30 &&
        match.HomeHistoricalGoals < 1.30 &&
        match.AwayHistoricalGoals < 1.30;

    // STEP 2: Balance Check
    double winDiff = Math.Abs(match.HomeWinPercent - match.AwayWinPercent);
    bool balancedMatch = winDiff <= 15.0;

    // STEP 3: Goal Environment
    bool validGoalEnvironment =
        match.Over1GoalsPercent >= 25.0 &&
        match.Over2GoalsPercent <= 18.0;

    // STEP 4: Exclusion Filters
    bool dominanceRisk =
        match.HomeAvgGoals >= 1.60 ||
        match.AwayAvgGoals >= 1.60 ||
        match.HomeHistoricalGoals >= 1.60 ||
        match.AwayHistoricalGoals >= 1.60 ||
        match.Over2GoalsPercent >= 25.0 ||
        winDiff >= 20.0;

    // FINAL DECISION
    return lowAverageGoals &&
           balancedMatch &&
           validGoalEnvironment &&
           !dominanceRisk;
}
public static double CalculateBTTSConfidence(MatchData match)
{
    double avgGoalScore =
        (match.HomeAvgGoals +
         match.AwayAvgGoals +
         match.HomeHistoricalGoals +
         match.AwayHistoricalGoals) / 4.0;

    double balanceScore =
        1.0 - (Math.Abs(match.HomeWinPercent - match.AwayWinPercent) / 100.0);

    double over1Score = match.Over1GoalsPercent / 100.0;

    // Weighted confidence (tuned for your data)
    double confidence =
        (avgGoalScore * 0.45) +
        (balanceScore * 0.35) +
        (over1Score * 0.20);

    return Math.Round(confidence * 100, 2); // percentage
}
